<script>
const $ = id => document.getElementById(id);
const exprEl = $("expression"), resEl = $("result");
const histPanel = $("history"), histList = $("historyList");
const OPS = ["+", "−", "×", "÷", "^"];

let expr = "", history = [], done = false, deg = true, ans = 0, lastGood = "0";

/* ---------- Safe expression parser (no eval) ---------- */
const bad = m => { throw new Error(m); };
const FN = ["asin","acos","atan","sin","cos","tan","ln","log","abs","exp","√","∛"];

function fact(n) {
  if (!Number.isInteger(n) || n < 0 || n > 170) bad("Math error");
  let r = 1;
  for (let i = 2; i <= n; i++) r *= i;
  return r;
}

function applyFn(f, v) {
  const k = deg ? Math.PI / 180 : 1;
  const clean = x => Math.abs(x) < 1e-12 ? 0 : x;
  switch (f) {
    case "sin": return clean(Math.sin(v * k));
    case "cos": return clean(Math.cos(v * k));
    case "tan":
      if (deg ? Math.abs(v % 180) === 90 : Math.abs(Math.cos(v)) < 1e-12) bad("Math error");
      return clean(Math.tan(v * k));
    case "asin": if (Math.abs(v) > 1) bad("Math error"); return Math.asin(v) / k;
    case "acos": if (Math.abs(v) > 1) bad("Math error"); return Math.acos(v) / k;
    case "atan": return Math.atan(v) / k;
    case "ln":   if (v <= 0) bad("Math error"); return Math.log(v);
    case "log":  if (v <= 0) bad("Math error"); return Math.log10(v);
    case "√":    if (v < 0) bad("Math error"); return Math.sqrt(v);
    case "∛":    return Math.cbrt(v);
    case "abs":  return Math.abs(v);
    case "exp":  return Math.exp(v);
  }
}

function evaluate(src) {
  const T = [];
  const re = /\s*(\d+\.?\d*|\.\d+|asin|acos|atan|sin|cos|tan|ln|log|abs|exp|Ans|π|e|√|∛|[-+−×÷*\/^()!%])/y;
  let p = 0;
  src = src.trim();
  while (p < src.length) {
    re.lastIndex = p;
    const m = re.exec(src);
    if (!m) bad("Error");
    T.push(m[1]);
    p = re.lastIndex;
  }
  let open = 0;
  T.forEach(t => { if (t === "(") open++; if (t === ")") open--; });
  if (open < 0) bad("Error");
  while (open-- > 0) T.push(")");

  let i = 0;
  const peek = () => T[i], next = () => T[i++];
  const isNum = t => t && /^[\d.]/.test(t);
  const isConst = t => t === "π" || t === "e" || t === "Ans";
  const expect = c => { if (next() !== c) bad("Error"); };

  function sum() {
    let v = term();
    while (peek() === "+" || peek() === "-" || peek() === "−") {
      const o = next(), r = term();
      v = o === "+" ? v + r : v - r;
    }
    return v;
  }
  function term() {
    let v = unary();
    for (;;) {
      const t = peek();
      if (t === "*" || t === "×") { next(); v *= unary(); }
      else if (t === "/" || t === "÷") { next(); const r = unary(); if (r === 0) bad("Can't divide by 0"); v /= r; }
      else if (t === "(" || isNum(t) || isConst(t) || FN.includes(t)) { v *= unary(); }
      else return v;
    }
  }
  function unary() {
    const t = peek();
    if (t === "-" || t === "−") { next(); return -unary(); }
    if (t === "+") { next(); return unary(); }
    return power();
  }
  function power() {
    const b = postfix();
    if (peek() === "^") { next(); return Math.pow(b, unary()); }
    return b;
  }
  function postfix() {
    let v = primary();
    while (peek() === "!" || peek() === "%") {
      v = next() === "!" ? fact(v) : v / 100;
    }
    return v;
  }
  function primary() {
    const t = next();
    if (t === undefined) bad("Error");
    if (isNum(t)) return parseFloat(t);
    if (t === "π") return Math.PI;
    if (t === "e") return Math.E;
    if (t === "Ans") return ans;
    if (t === "(") { const v = sum(); expect(")"); return v; }
    if (FN.includes(t)) { expect("("); const v = sum(); expect(")"); return applyFn(t, v); }
    bad("Error");
  }

  const result = sum();
  if (i < T.length) bad("Error");
  if (!Number.isFinite(result)) bad("Math error");
  return result;
}

/* ---------- Formatting ---------- */
function fmt(n) {
  if (Math.abs(n) < 1e-12) n = 0;
  n = Number(n.toPrecision(12));
  const a = Math.abs(n);
  if (a >= 1e15 || (a > 0 && a < 1e-7)) return n.toExponential(6).replace(/\.?0+e/, "e");
  return n.toLocaleString("en-US", { maximumFractionDigits: 10 });
}
function raw(n) {
  if (Math.abs(n) < 1e-12) n = 0;
  return String(Number(n.toPrecision(12))).replace(/e\+?(-?\d+)/, "×10^($1)").replace(/^-/, "−");
}

/* ---------- Display ---------- */
function size() {
  const n = resEl.textContent.length;
  resEl.style.fontSize = n > 14 ? "28px" : n > 10 ? "36px" : "46px";
}
function update() {
  exprEl.textContent = expr;
  exprEl.scrollLeft = exprEl.scrollWidth;
  const s = expr.replace(/[+−×÷^]+$/, "");
  if (!s || s === "−") { resEl.textContent = "0"; lastGood = "0"; return size(); }
  try { lastGood = fmt(evaluate(s)); } catch (e) {}
  resEl.textContent = lastGood;
  size();
}

/* ---------- Input ---------- */
function equals() {
  if (!expr) return;
  const s = expr.replace(/[+−×÷^]+$/, "");
  try {
    const v = evaluate(s);
    history.unshift({ e: s, v });
    if (history.length > 30) history.pop();
    ans = v;
    exprEl.textContent = s + " =";
    expr = raw(v);
    resEl.textContent = fmt(v);
    size();
    done = true;
  } catch (e) {
    resEl.textContent = e.message || "Error";
    size();
  }
}

function press(k) {
  if (k === "AC") { expr = ""; done = false; return update(); }
  if (k === "BACK") {
    if (done) { expr = ""; done = false; }
    else expr = expr.replace(/(asin\(|acos\(|atan\(|sin\(|cos\(|tan\(|ln\(|log\(|abs\(|exp\(|√\(|∛\(|Ans|[\s\S])$/u, "");
    return update();
  }
  if (k === "=") return equals();

  const isOp = OPS.includes(k);
  const isPost = k === "!" || k === "%" || k === "^2";
  if (done) { if (!(isOp || isPost)) expr = ""; done = false; }

  if (isOp) {
    if (expr === "") { if (k === "−") expr = "−"; }
    else if (expr === "−") { /* ignore */ }
    else {
      const last = expr.slice(-1);
      if (OPS.includes(last)) expr = expr.slice(0, -1) + k;
      else if (last === "(") { if (k === "−") expr += k; }
      else expr += k;
    }
  } else if (isPost) {
    if (/[\d)!%πe]$|Ans$/.test(expr)) expr += k;
  } else if (k === ".") {
    const num = expr.match(/[\d.]*$/)[0];
    if (!num.includes(".")) {
      if (num === "") expr += /[)πe!%]$|Ans$/.test(expr) ? "×0." : "0.";
      else expr += ".";
    }
  } else if (k === ")") {
    const o = (expr.match(/\(/g) || []).length, c = (expr.match(/\)/g) || []).length;
    if (o > c && !/[(+−×÷^]$/.test(expr)) expr += ")";
  } else {
    expr += k;
  }
  update();
}

/* ---------- Events ---------- */
document.querySelector(".calculator").addEventListener("click", e => {
  const b = e.target.closest("[data-k]");
  if (b) press(b.dataset.k);
});

document.addEventListener("keydown", e => {
  if (e.ctrlKey || e.metaKey || e.altKey) return;
  const map = { "*": "×", "/": "÷", "-": "−", "Enter": "=", "Backspace": "BACK", "Escape": "AC", "p": "π" };
  const k = map[e.key] || e.key;
  if (/^[0-9]$/.test(k) || ["+","−","×","÷","^",".","!","%","(",")","=","BACK","AC","π"].includes(k)) {
    e.preventDefault();
    press(k);
  }
});

$("modeBtn").addEventListener("click", function () {
  deg = !deg;
  this.textContent = deg ? "DEG" : "RAD";
  update();
});
$("sciBtn").addEventListener("click", function () {
  $("sci").classList.toggle("hide");
  this.classList.toggle("on");
});
$("themeBtn").addEventListener("click", function () {
  const root = document.documentElement;
  const light = root.getAttribute("data-theme") === "light";
  if (light) { root.removeAttribute("data-theme"); this.textContent = "☀"; }
  else { root.setAttribute("data-theme", "light"); this.textContent = "☾"; }
});

/* ---------- History ---------- */
function renderHistory() {
  histList.innerHTML = "";
  if (!history.length) {
    const li = document.createElement("li");
    li.className = "empty";
    li.textContent = "No calculations yet";
    histList.appendChild(li);
    return;
  }
  history.forEach(item => {
    const li = document.createElement("li");
    li.className = "history-item";
    const a = document.createElement("div");
    a.className = "history-expression";
    a.textContent = item.e + " =";
    const b = document.createElement("div");
    b.className = "history-result";
    b.textContent = fmt(item.v);
    li.append(a, b);
    li.addEventListener("click", () => {
      expr = raw(item.v);
      done = true;
      histPanel.classList.remove("open");
      exprEl.textContent = item.e + " =";
      resEl.textContent = fmt(item.v);
      size();
    });
    histList.appendChild(li);
  });
}
$("historyBtn").addEventListener("click", () => { renderHistory(); histPanel.classList.add("open"); });
$("closeHistory").addEventListener("click", () => histPanel.classList.remove("open"));
$("clearHistory").addEventListener("click", () => { history = []; renderHistory(); });

renderHistory();
update();
</script>
